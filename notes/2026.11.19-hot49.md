|---|---|---|
| 语言 | C 和 C++ 都有 | 只有 C++ |
| 本质 | 编译期运算符 | 运行时成员函数 |

| 对 vector | 对象自身字节数（无意义） | 元素个数 ✅ |

# 题解

## 方法一(统计字母出现次数)
```cpp
class Solution {
public:
    vector<vector<string>> groupAnagrams(vector<string>& strs) {
        unordered_map<string,vector<string>> mp;
        for(auto& str:strs){
            int hash[26] = {0};
            
            for(int i = 0;i < str.size();i++){
                hash[str[i] - 'a']++;  // 统计每个单词中每个字母出现的次数
            }
            string key;
            for(int j = 0;j < 26;j++){
                key += hash[j];
                key += "#";
            }
            mp[key].emplace_back(str);
        }
        vector<vector<string>> ans;
        for(auto &p : mp){
            ans.push_back(p.second);
        }
        return ans;
    }
};
```

        unordered_map<string,vector<string>> mp;
        for(auto &str: strs){ // 遍历每个单词，对某个单词进行操作
            string key = str;
            sort(key.begin(),key.end()); // 字母正序排列
            mp[key].push_back(str);
        }
        vector<vector<string>> ans;
        for(auto &p : mp){
            ans.push_back(p.second);
        }
        return ans;
    }
};

class Solution {
public:
    vector<vector<string>> groupAnagrams(vector<string>& strs) {
        unordered_map<string,vector<string>> mp;
        for(auto & str : strs){
            int hash[26] = {0};
            for(auto & c : str){
                hash[c - 'a']++;
            }
            int i = 0;
            string key = "";
            while(i < 26){
                key += to_string(hash[i]); // 对string追加元素
                key += "#";
                i++;
            }
            mp[key].push_back(str);
        }
        vector<vector<string>> res;
        for(auto & p : mp){
            res.push_back(p.second);
        }
        return res;

    }
};
